const axios = require("axios");

const mahmud = async () => {
  const base = await axios.get("https://raw.githubusercontent.com/mahmudx7/HINATA/main/baseApiUrl.json");
  return base.data.mahmud;
};

module.exports = {
    config: {
        name: "اغنية2",
        version: "1.8",
        author: "MahMUD", 
        countDown: 5,
        role: 0,
        category: "music",
        guide: {
            ar: "{p}sing2 [اسم الأغنية]"
        }
    },

    onStart: async function ({ api, event, args, message }) {
        if (args.length === 0) {
            return message.reply("❌ | يرجى كتابة اسم الأغنية التي تريد البحث عنها!\n\nمثال: sing2 تشيما");
        }

        try {
            const query = encodeURIComponent(args.join(" "));
            const apiUrl = `${await mahmud()}/api/sing?q=${query}`;

            const response = await axios.get(apiUrl, {
                responseType: "stream",
                headers: { "author": module.exports.config.author }
            });

            if (response.data && response.data.error) {
                return message.reply(`❌ | ${response.data.error}`);
            }

            return message.reply({
                body: `✅ | تفضل أغنيتك المطلوبة: ${args.join(" ")} 🎵`,
                attachment: response.data
            });

        } catch (error) {
            console.error("Error:", error.message);

            if (error.response) {
                return message.reply(`❌ | حدث خطأ في الخادم: ${error.message}`);
            }

            return message.reply("🥺 عذراً، حدث خطأ أثناء تحميل الأغنية، حاول مرة أخرى.");
        }
    }
};
